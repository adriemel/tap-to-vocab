Add:
```javascript
skipBtn.addEventListener('click', function () {
  if (isAnswering) return;
  isAnswering = true;
  btn1.disabled = true;
  btn2.disabled = true;
  skipBtn.disabled = true;
  window.speechSynthesis.cancel();
  var done = false;
  function go() {
    if (done) return;
    done = true;
    setTimeout(function () {
      currentIndex++;
      if (currentIndex >= questions.length) {
        showEndScreen();
      } else {
        showQuestion(currentIndex);
      }
      isAnswering = false;
    }, 200);
  }
  go();
});
```
