    /* ── Bubble rendering ── */
    function appendBubble(text, side) {
      // side: 'left' (question) or 'right' (answer)
      var row = document.createElement('div');
      row.className = 'qs-bubble-row ' + side;

      var bubble = document.createElement('div');
      bubble.className = 'qs-bubble ' + (side === 'left' ? 'question' : 'answer');
      bubble.textContent = text;

      row.appendChild(bubble);
      chatEl.appendChild(row);

      // CHAT-06: scroll latest bubble into view
      bubble.scrollIntoView({ behavior: 'smooth', block: 'end' });
      return bubble;
    }

    /* ── Show question N (0-indexed) ── */
    function showQuestion(i) {
      currentIndex = i;
      var q = questions[i];

      // Render question bubble (CHAT-02)
      appendBubble(q.question, 'left');

      // AUDIO-01: speak question on appear
      speakSpanish(q.question);

      // Update progress pill (e.g., "3 / 14")
      progressEl.textContent = (i + 1) + ' / ' + questions.length;

      // CHAT-03: render two choice buttons with short labels
      btn1.textContent = q.label1;
      btn2.textContent = q.label2;
      // Accessibility: aria-label uses full answer text (per UI-SPEC Accessibility section)
      btn1.setAttribute('aria-label', q.answer1);
      btn2.setAttribute('aria-label', q.answer2);
      btn1.disabled = false;
      btn2.disabled = false;
    }

    /* ── Handle choice tap ── */
    function onChoiceTap(choiceIndex) {
      // Pitfall 2: guard against double-tap during 1200ms transition
      if (isAnswering) return;
      isAnswering = true;
      btn1.disabled = true;
      btn2.disabled = true;

      var q = questions[currentIndex];
      var answerText = (choiceIndex === 0) ? q.answer1 : q.answer2;
      chosenAnswers.push(answerText);

      // CHAT-04: append right-aligned answer bubble (also triggers CHAT-06 scroll)
      appendBubble(answerText, 'right');

      // AUDIO-02: speak chosen answer 400ms after it lands
      setTimeout(function () { speakSpanish(answerText); }, 400);

      // CHAT-05: 1200ms after answer lands, advance to next question OR end screen
      setTimeout(function () {
        currentIndex++;
        if (currentIndex >= questions.length) {
          showEndScreen();
        } else {
          showQuestion(currentIndex);
        }
        isAnswering = false;
      }, 1200);
    }

    btn1.addEventListener('click', function () { onChoiceTap(0); });
    btn2.addEventListener('click', function () { onChoiceTap(1); });

    /* ── End screen ── */
    function buildParagraph() {
      // END-02: chosen answers joined with single space + closing line
      return chosenAnswers.join(' ') + ' ¡Muchas gracias por escuchar!';

    function showEndScreen() {
      // END-01: hard swap (no transition) per UI-SPEC
      pageEl.style.display = 'none';
      endEl.classList.add('visible');
      endEl.setAttribute('aria-hidden', 'false');

      var paragraph = buildParagraph();
      introTextEl.textContent = paragraph;

      // END-03: auto-read paragraph 300ms after end screen paint
      setTimeout(function () { speakSpanish(paragraph); }, 300);
    }

    /* ── End screen actions ── */
    // END-04: replay re-reads the paragraph
    replayBtn.addEventListener('click', function () {
      speakSpanish(buildParagraph());
    });

    // END-05: start again resets state, hides end screen, shows chat with question 1

      window.speechSynthesis.cancel();
      currentIndex = 0;

      endEl.classList.remove('visible');

    });

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Static file server → browser fetch | `data/quien-soy-sentences.txt` is served same-origin as plain text. The file is developer-controlled, committed to git, and contains only conversational Spanish text. No untrusted input crosses this boundary. |
| User → DOM | User can tap one of two pre-defined choice buttons. There is no text input field, no form, no upload. Choice index (0 or 1) is the only user-controlled value, and it is range-checked by being the only two options exposed. |
| TTS API → audio output | `SpeechSynthesisUtterance(text)` is a browser-native, sandboxed API. Text passed to it is rendered as audio; the API does not execute the text. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-19-06 | Spoofing | Inicio anchor `<a href="/">` | accept | Same-origin link to root; no auth context, no token leakage. ASVS V4 not applicable. |
| T-19-07 | Tampering | data/quien-soy-sentences.txt content | mitigate | The IIFE uses `bubble.textContent = text` (not `innerHTML`) when rendering bubbles. Even if the data file is tampered to include HTML/JS markup, the browser will render it as literal text characters, not parse it. ASVS V5.3.3 (output encoding) — IMPLEMENTED via `textContent`. |
| T-19-08 | Repudiation | N/A | n/a | No actions are recorded or attributable; pure local practice flow. |
| T-19-09 | Information Disclosure | localStorage / sessionStorage | accept | This phase does not read or write any storage. No PII, no secrets exposed. |
| T-19-10 | Denial of Service | TTS queue | mitigate | `speakSpanish` calls `speechSynthesis.cancel()` before each utterance, preventing queue buildup. The `setTimeout(100)` wrapper ensures iOS Safari completes the cancel before queueing the next utterance (Pitfall 3 in RESEARCH). |
| T-19-11 | Denial of Service | Double-tap advancing index past array bounds | mitigate | `isAnswering` boolean guard at the top of `onChoiceTap` prevents re-entry during the 1200ms transition. |
| T-19-12 | Elevation of Privilege | N/A | n/a | No auth, no roles, no privileged actions. ASVS V4 not applicable. |
| T-19-13 | Tampering | URL parameters / query string | accept | `quien-soy.html` does not read any URL parameters; nothing to tamper with. The page state is closure-scoped in the IIFE. |

**Severity assessment:** All threats are LOW or fully mitigated. T-19-07 (Tampering — XSS via data file) is the highest-impact theoretical concern; mitigated by `textContent` usage. T-19-10 and T-19-11 are availability concerns mitigated by the `cancel()` and `isAnswering` patterns respectively. No HIGH-severity threats per ASVS L1 — phase 19 has no authentication, no server-side processing, no free-form user input, and no data persistence (per RESEARCH.md Security Domain section).
